4. 4. Traitement du formulaire d'ajout
<?php
if (isset($_POST['ajouter'])) {
    $nom = $_POST['nom'];
    $prix = $_POST['prix'];
    $quantite = $_POST['quantite'];

    // Requête pour insérer un nouveau produit dans la base de données
    $sql = "INSERT INTO produits (nom, prix, quantite) VALUES ('$nom', '$prix', '$quantite')";

    if ($conn->query($sql) === TRUE) {
        echo "Nouveau produit ajouté avec succès.";
    } else {
        echo "Erreur : " . $sql . "<br>" . $conn->error;
    }
}
?>




5. Modification d'un produit (écran d'édition)
<?php
// Connexion à la base de données
include('db_connect.php');  // Si tu utilises un fichier séparé pour la connexion

if (isset($_GET['id'])) {
    $id = $_GET['id'];
    
    // Requête pour récupérer le produit à modifier
    $sql = "SELECT * FROM produits WHERE id = '$id'";
    $result = $conn->query($sql);
    $product = $result->fetch_assoc();
}
?>

<h2>Modifier le produit</h2>

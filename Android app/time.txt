}

<p id="chronometer">00:00:00</p> <!-- Le chronomètre sera mis à jour ici -->

<script>
        // Fonction pour calculer et afficher le temps écoulé
        function updateChronometer() {
            var startTime = <?php echo $_SESSION['start_time']; ?>;  // Récupérer l'heure de départ de la session
            var currentTime = Math.floor(Date.now() / 1000); // Temps actuel en secondes

            // Calcul du temps écoulé
            var elapsed = currentTime - startTime;

            // Conversion du temps en heures, minutes, secondes
            var hours = Math.floor(elapsed / 3600);
            var minutes = Math.floor((elapsed % 3600) / 60);
            var seconds = elapsed % 60;

            // Formatage avec 2 chiffres pour les minutes et secondes
            minutes = minutes < 10 ? "0" + minutes : minutes;
            seconds = seconds < 10 ? "0" + seconds : seconds;

            // Affichage du chronomètre
            document.getElementById("chronometer").innerText = hours + ":" + minutes + ":" + seconds;
        }

        // Mettre à jour le chronomètre toutes les secondes
        setInterval(updateChronometer, 1000);
    </script>
